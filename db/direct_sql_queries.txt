-- Avg. temperature across stations for past week of current year compared with the same period of historic averages
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT
    TO_CHAR(ts, 'MM-DD') AS day_of_year,
    ROUND(AVG(temperature_c)::numeric, 2) AS historic_avg_all_years,
    ROUND(AVG(CASE WHEN EXTRACT(YEAR FROM ts) = EXTRACT(YEAR FROM CURRENT_DATE) THEN temperature_c END)::numeric, 2) AS avg_current_year
FROM weather.hyper
WHERE (EXTRACT(MONTH FROM ts), EXTRACT(DAY FROM ts)) IN (
    SELECT EXTRACT(MONTH FROM d), EXTRACT(DAY FROM d)
    FROM generate_series(CURRENT_DATE - INTERVAL '7 days', CURRENT_DATE - INTERVAL '1 day', '1 day') AS d
)
GROUP BY TO_CHAR(ts, 'MM-DD')
ORDER BY day_of_year;
"

-- Combined air + weather snapshot
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "SELECT
  COALESCE(a.fiwareid, w.fiwareid) AS fiwareid,
  COALESCE(a.ts, w.ts) AS ts,
  a.no2, a.o3, a.so2, a.co, a.pm10, a.pm25,
  a.air_quality_summary,
  w.temperature_c, w.humidity_pct, w.pressure_hpa,
  w.wind_dir_deg, w.wind_speed_ms, w.precip_mm,
  COALESCE(a.lat, w.lat) AS lat,
  COALESCE(a.lon, w.lon) AS lon
FROM air.latest a
FULL OUTER JOIN weather.latest w USING (fiwareid);"

-- Air Quality Index (AQI) categorization
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "SELECT
  fiwareid,
  ts,
  pm25, pm10, no2, o3, so2, co,
  CASE
    WHEN pm25 IS NULL AND pm10 IS NULL THEN 'No PM Data'
    WHEN COALESCE(pm25, 0) <= 12 AND COALESCE(pm10, 0) <= 54 THEN 'Good'
    WHEN COALESCE(pm25, 0) <= 35.4 OR COALESCE(pm10, 0) <= 154 THEN 'Moderate'
    WHEN COALESCE(pm25, 0) <= 55.4 OR COALESCE(pm10, 0) <= 254 THEN 'Unhealthy for Sensitive Groups'
    WHEN COALESCE(pm25, 0) <= 150.4 OR COALESCE(pm10, 0) <= 354 THEN 'Unhealthy'
    WHEN COALESCE(pm25, 0) <= 250.4 OR COALESCE(pm10, 0) <= 424 THEN 'Very Unhealthy'
    ELSE 'Hazardous'
  END AS aqi_category,
  lat, lon
FROM air.latest;"

-- Station inventory/metadata
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "SELECT * FROM (SELECT fiwareid, 'air' AS data_type, MIN(ts) AS first_seen, MAX(ts) AS last_seen, COUNT(*) AS record_count, AVG(lat) AS lat, AVG(lon) AS lon FROM air.hyper GROUP BY fiwareid UNION ALL SELECT fiwareid, 'weather' AS data_type, MIN(ts) AS first_seen, MAX(ts) AS last_seen, COUNT(*) AS record_count, AVG(lat) AS lat, AVG(lon) AS lon FROM weather.hyper GROUP BY fiwareid) t ORDER BY fiwareid;"

-- Daily trends (for charting)

docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "SELECT

  bucket_day::date AS day,
  temp_avg_c, humidity_avg, pressure_avg, precip_max_mm,

  temp_avg_c - LAG(temp_avg_c) OVER (PARTITION BY fiwareid ORDER BY bucket_day) AS temp_delta

-- Data freshness monitoring
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "SELECT
  'air' AS schema_name,
  fiwareid,
  MAX(ts) AS last_update,
  NOW() - MAX(ts) AS staleness,
  CASE
    WHEN NOW() - MAX(ts) < INTERVAL '1 hour' THEN 'Fresh'
    WHEN NOW() - MAX(ts) < INTERVAL '6 hours' THEN 'Stale'
    ELSE 'Offline'
  END AS status
FROM air.hyper
GROUP BY fiwareid
UNION ALL
SELECT
  'weather' AS schema_name,
  fiwareid,
  MAX(ts) AS last_update,
  NOW() - MAX(ts) AS staleness,
  CASE
    WHEN NOW() - MAX(ts) < INTERVAL '1 hour' THEN 'Fresh'
    WHEN NOW() - MAX(ts) < INTERVAL '6 hours' THEN 'Stale'
    ELSE 'Offline'
  END AS status
FROM weather.hyper
GROUP BY fiwareid
ORDER BY fiwareid;"

-- ==============================================
-- VLC Data Quality Checks
-- ==============================================
-- 1. NULLS - Critical fields that should never be NULL
-- Primary key fields must have values
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT *
FROM air.hyper
WHERE fiwareid IS NULL
   OR ts IS NULL;"

docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT *
FROM weather.hyper
WHERE fiwareid IS NULL
   OR ts IS NULL;"

-- 2. UNIQUENESS - Duplicate primary key combinations (fiwareid + ts)
-- Should return 0 rows if PK constraint is enforced
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT fiwareid, ts, COUNT(*)
FROM air.hyper
GROUP BY fiwareid, ts
HAVING COUNT(*) > 1;"

docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT fiwareid, ts, COUNT(*)
FROM weather.hyper
GROUP BY fiwareid, ts
HAVING COUNT(*) > 1;"

-- 3. INTEGRITY - Unknown fiwareid values (orphan station IDs)
-- Cross-check with expected station list
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT DISTINCT fiwareid
FROM air.hyper
WHERE fiwareid NOT IN (
    SELECT DISTINCT fiwareid FROM weather.hyper
);"

docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT DISTINCT fiwareid
FROM weather.hyper
WHERE fiwareid NOT IN (
    SELECT DISTINCT fiwareid FROM air.hyper
);"

-- 4. ACCEPTED VALUES - Air quality summary validation
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT *
FROM air.hyper
WHERE air_quality_summary IS NOT NULL
  AND air_quality_summary NOT IN ('Buena', 'Admisible', 'Mejorable', 'Mala', 'Muy mala', 'Razonablemente Buena', 'Hazardous', 'Desfavorable', 'Regular');"

-- 5. RANGE - Physical measurement bounds
-- Air quality out-of-range values
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT *
FROM air.hyper
WHERE no2 < 0 OR no2 > 1000
   OR o3 < 0 OR o3 > 500
   OR so2 < 0 OR so2 > 2000
   OR co < 0 OR co > 100
   OR pm10 < 0 OR pm10 > 1000
   OR pm25 < 0 OR pm25 > 500;"

-- Weather out-of-range values
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT *
FROM weather.hyper
WHERE temperature_c < -40 OR temperature_c > 60
   OR humidity_pct < 0 OR humidity_pct > 105
   OR pressure_hpa < 800 OR pressure_hpa > 1100
   OR wind_speed_ms < 0 OR wind_speed_ms > 100
   OR wind_dir_deg < 0 OR wind_dir_deg > 360
   OR precip_mm < 0 OR precip_mm > 500;"

-- 6. GEO RANGE - Latitude/Longitude must be in Valencia area

FROM air.hyper

-- 9. NULL SPIKE - More than 30% NULLs in measurement columns
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT
    'air.hyper' AS table_name,
    COUNT(*) AS total_rows,
    ROUND(100.0 * COUNT(*) FILTER (WHERE no2 IS NULL) / NULLIF(COUNT(*), 0), 2) AS no2_null_pct,
    ROUND(100.0 * COUNT(*) FILTER (WHERE o3 IS NULL) / NULLIF(COUNT(*), 0), 2) AS o3_null_pct,
    ROUND(100.0 * COUNT(*) FILTER (WHERE pm10 IS NULL) / NULLIF(COUNT(*), 0), 2) AS pm10_null_pct,
    ROUND(100.0 * COUNT(*) FILTER (WHERE pm25 IS NULL) / NULLIF(COUNT(*), 0), 2) AS pm25_null_pct
FROM air.hyper;"

docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT
    'weather.hyper' AS table_name,
    COUNT(*) AS total_rows,
    ROUND(100.0 * COUNT(*) FILTER (WHERE temperature_c IS NULL) / NULLIF(COUNT(*), 0), 2) AS temp_null_pct,
    ROUND(100.0 * COUNT(*) FILTER (WHERE humidity_pct IS NULL) / NULLIF(COUNT(*), 0), 2) AS humidity_null_pct,
    ROUND(100.0 * COUNT(*) FILTER (WHERE pressure_hpa IS NULL) / NULLIF(COUNT(*), 0), 2) AS pressure_null_pct

-- ==============================================
-- VLC Hourly averages
-- ==============================================
-- (i) Hourly averages across all stations since everything started
-- AIR
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT
  time_bucket('1 hour', ts) AS bucket_hour,
  AVG(no2) AS no2_avg,
  AVG(o3) AS o3_avg,
  AVG(so2) AS so2_avg,
  AVG(co) AS co_avg,
  AVG(pm10) AS pm10_avg,
  AVG(pm25) AS pm25_avg
FROM air.hyper
GROUP BY bucket_hour
ORDER BY bucket_hour;"

-- WEATHER
docker compose -f compose/docker-compose.yml exec -T timescaledb psql -U vlc_dev -d vlc -c "
SELECT
  time_bucket('1 hour', ts) AS bucket_hour,
  AVG(temperature_c) AS temp_avg_c,
  AVG(humidity_pct) AS humidity_avg,
  AVG(pressure_hpa) AS pressure_avg,
  AVG(wind_speed_ms) AS wind_avg_ms,
  degrees(atan2(avg(sin(radians(wind_dir_deg))), avg(cos(radians(wind_dir_deg))))) AS wind_avg_deg,
  MAX(precip_mm) AS precip_max_mm
FROM weather.hyper
GROUP BY bucket_hour
ORDER BY bucket_hour;"

-- AIR
